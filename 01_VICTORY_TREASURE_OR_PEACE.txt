<!-- LOCAL VICTORY THEME: self-contained Web Audio; does not use audio.js -->
<script>
(() => {
    "use strict";

    // This snippet listens only for clicks on controls whose visible text matches
    // this ending choice. Keep it in the HTML file containing that choice.
    const endingChoicePattern = /claim\s+(?:the\s+)?treasure|treasure\s+claim|peace(?:\s+restored)?\s+ending|restore\s+peace|choose\s+peace/i;
    const navigationDelayMs = 3600;
    let isPlaying = false;
    let lockedUntil = 0;

    function playEndingTheme() {
        if (isPlaying || Date.now() < lockedUntil) return false;
        const AudioContextClass = window.AudioContext || window.webkitAudioContext;
        if (!AudioContextClass) return;

        let context;
        try { context = new AudioContextClass(); }
        catch (_) { return; }

        isPlaying = true;
        lockedUntil = Date.now() + navigationDelayMs;
        if (context.state === "suspended") context.resume().catch(() => {});

        const master = context.createGain();
        master.gain.setValueAtTime(0.72, context.currentTime);
        master.connect(context.destination);
        const start = context.currentTime + 0.035;

        function note(freq, offset, length, volume = 0.12, wave = "sine", detune = 0) {
            const oscillator = context.createOscillator();
            const envelope = context.createGain();
            const when = start + offset;
            const attack = Math.min(0.12, Math.max(0.015, length * 0.16));
            oscillator.type = wave;
            oscillator.frequency.setValueAtTime(freq, when);
            if (detune) oscillator.detune.setValueAtTime(detune, when);
            envelope.gain.setValueAtTime(0.0001, when);
            envelope.gain.exponentialRampToValueAtTime(Math.max(0.0002, volume), when + attack);
            envelope.gain.setTargetAtTime(0.0001, when + Math.max(attack + 0.03, length * 0.68), Math.max(0.06, length * 0.16));
            oscillator.connect(envelope);
            envelope.connect(master);
            oscillator.start(when);
            oscillator.stop(when + length + 0.18);
        }

        try {
            // Warm, triumphant rising arpeggio over a soft major chord.
            note(130.81, 0.00, 3.10, 0.045, "sine"); // C3
            note(196.00, 0.00, 3.10, 0.035, "sine"); // G3
            note(261.63, 0.00, 2.80, 0.040, "sine"); // C4
            note(329.63, 0.03, 2.65, 0.030, "sine"); // E4
            const melody = [
                [523.25, 0.10, 0.34, 0.115], [659.25, 0.43, 0.34, 0.105],
                [783.99, 0.76, 0.42, 0.105], [1046.50, 1.15, 0.72, 0.105],
                [783.99, 1.72, 0.28, 0.075], [1046.50, 2.00, 0.34, 0.095],
                [1318.51, 2.33, 0.52, 0.075], [1046.50, 2.75, 0.72, 0.065]
            ];
            melody.forEach(([freq, at, len, vol]) => note(freq, at, len, vol, "sine"));
            note(1567.98, 1.28, 0.45, 0.035, "sine");
            note(2093.00, 2.38, 0.55, 0.025, "sine");
        } catch (error) {
            console.warn("Local ending theme could not be played.", error);
            isPlaying = false;
        }

        window.setTimeout(() => {
            context.close().catch(() => {});
            isPlaying = false;
        }, 4100);
    }

    document.addEventListener("click", (event) => {
        const origin = event.target instanceof Element ? event.target : event.target?.parentElement;
        const control = origin?.closest("a[href], button, [role='button'], input[type='button'], input[type='submit']");
        if (!control || control.disabled) return;

        const label = [
            control.innerText,
            control.textContent,
            control.value,
            control.getAttribute("aria-label"),
            control.getAttribute("title")
        ].filter(Boolean).join(" ").replace(/\s+/g, " ").trim();
        if (!endingChoicePattern.test(label)) return;

        const rawHref = control.matches("a[href]") ? control.getAttribute("href") : "";
        const shouldDelayNavigation = Boolean(
            rawHref &&
            !rawHref.startsWith("#") &&
            !rawHref.toLowerCase().startsWith("javascript:") &&
            !control.hasAttribute("download") &&
            control.target !== "_blank"
        );

        if (shouldDelayNavigation) {
            // Start within the trusted click, then delay a normal link navigation
            // only if Web Audio actually initialized.
            const destination = control.href;
            const didStart = playEndingTheme();
            if (didStart) {
                event.preventDefault();
                window.setTimeout(() => window.location.assign(destination), navigationDelayMs);
            }
        } else {
            playEndingTheme();
        }
    }, true);
})();
</script>
